"""JSON serialization for scenarios and game weekends.

Times are stored as the wall time and IANA zone they were *entered* in, not as
a normalised UTC stamp. That is what makes the round-trip lossless in the sense
the spec asks for: a flight entered as departing 21:30 Denver and arriving
23:40 Chicago comes back the same way, not as two Chicago timestamps.

Fields added after the first format (an event's kickoff anchor, a window's edge
sources, the Saturday-night assumed wake) are optional keys written only when
set, so files saved before they existed load unchanged.

A game-weekend file is different in kind: it stores what staff enter (the
template, kickoff, zones, assumptions, and overrides) and is rebuilt through
``build_weekend`` on load, so resolved instants are never trusted from disk.
"""

from __future__ import annotations

import json
from datetime import timedelta
from pathlib import Path
from typing import Any

from .schedule import AssumptionSet, Event, KickoffAnchor, ProtectedSleepWindow, Scenario
from .temporal import Interval, LocalTime
from .weekend import (
    GameWeekend,
    KickoffBandTemplate,
    NightMarker,
    TemplateEntry,
    WindowOverride,
    build_weekend,
)

__all__ = [
    "load_scenario",
    "load_weekend",
    "save_scenario",
    "save_weekend",
    "scenario_from_dict",
    "scenario_to_dict",
    "weekend_from_dict",
    "weekend_to_dict",
]

FORMAT_VERSION = 1
WEEKEND_KIND = "game-weekend"


def _interval_to_dict(interval: Interval) -> dict[str, Any]:
    payload: dict[str, Any] = {
        "start": interval.wall("start"),
        "end": interval.wall("end"),
        "start_zone": interval.start_zone,
        "end_zone": interval.end_zone,
    }
    if interval.start_fold is not None:
        payload["start_fold"] = interval.start_fold
    if interval.end_fold is not None:
        payload["end_fold"] = interval.end_fold
    return payload


def _interval_from_dict(payload: dict[str, Any]) -> Interval:
    return Interval.of(
        payload["start"],
        payload["end"],
        payload["start_zone"],
        end_zone=payload.get("end_zone"),
        start_fold=payload.get("start_fold"),
        end_fold=payload.get("end_fold"),
    )


def _minutes(delta: timedelta) -> int:
    seconds = delta.total_seconds()
    if seconds % 60:
        raise ValueError(f"offset {delta} is not a whole number of minutes and would be truncated")
    return int(seconds // 60)


def _anchor_to_dict(anchor: KickoffAnchor) -> dict[str, Any]:
    return {
        "start_minutes": _minutes(anchor.start),
        "end_minutes": _minutes(anchor.end),
        "start_zone": anchor.start_zone,
        "end_zone": anchor.end_zone,
    }


def _anchor_from_dict(payload: dict[str, Any]) -> KickoffAnchor:
    return KickoffAnchor(
        start=timedelta(minutes=payload["start_minutes"]),
        end=timedelta(minutes=payload["end_minutes"]),
        start_zone=payload["start_zone"],
        end_zone=payload.get("end_zone"),
    )


def event_to_dict(event: Event) -> dict[str, Any]:
    payload: dict[str, Any] = {
        "id": event.id,
        "kind": event.kind,
        "title": event.title,
        "movable": event.movable,
        "location": event.location,
        "interval": _interval_to_dict(event.interval),
    }
    if event.anchor is not None:
        payload["anchor"] = _anchor_to_dict(event.anchor)
    return payload


def event_from_dict(payload: dict[str, Any]) -> Event:
    raw_anchor = payload.get("anchor")
    return Event(
        id=payload["id"],
        kind=payload["kind"],
        title=payload["title"],
        interval=_interval_from_dict(payload["interval"]),
        movable=payload.get("movable", False),
        location=payload.get("location"),
        anchor=None if raw_anchor is None else _anchor_from_dict(raw_anchor),
    )


def assumptions_to_dict(assumptions: AssumptionSet) -> dict[str, Any]:
    payload: dict[str, Any] = {
        "sleep_target_hours": assumptions.sleep_target_hours,
        "pre_bed_minutes": assumptions.pre_bed_minutes,
        "morning_minutes": assumptions.morning_minutes,
        "source": assumptions.source,
    }
    if assumptions.saturday_night_wake is not None:
        payload["saturday_night_wake"] = assumptions.saturday_night_wake
    return payload


def assumptions_from_dict(payload: dict[str, Any]) -> AssumptionSet:
    return AssumptionSet(**payload)


def _window_to_dict(window: ProtectedSleepWindow) -> dict[str, Any]:
    payload: dict[str, Any] = {
        "id": window.id,
        "label": window.label,
        "kind": window.kind,
        "proposed": window.proposed,
        "interval": _interval_to_dict(window.interval),
    }
    if window.start_source is not None:
        payload["start_source"] = window.start_source
    if window.end_source is not None:
        payload["end_source"] = window.end_source
    return payload


def _window_from_dict(payload: dict[str, Any]) -> ProtectedSleepWindow:
    return ProtectedSleepWindow(
        id=payload["id"],
        label=payload["label"],
        interval=_interval_from_dict(payload["interval"]),
        kind=payload.get("kind", "hotel"),
        proposed=payload.get("proposed", False),
        start_source=payload.get("start_source"),
        end_source=payload.get("end_source"),
    )


def _check_version(payload: dict[str, Any], what: str) -> None:
    version = payload.get("format_version")
    if version != FORMAT_VERSION:
        raise ValueError(
            f"unsupported {what} format_version {version!r}; expected {FORMAT_VERSION}"
        )


def scenario_to_dict(scenario: Scenario) -> dict[str, Any]:
    assumptions = scenario.assumptions
    return {
        "format_version": FORMAT_VERSION,
        "id": scenario.id,
        "title": scenario.title,
        "display_zone": scenario.display_zone,
        "assumptions": None if assumptions is None else assumptions_to_dict(assumptions),
        "events": [event_to_dict(e) for e in scenario.events],
        "windows": [_window_to_dict(w) for w in scenario.windows],
    }


def scenario_from_dict(payload: dict[str, Any]) -> Scenario:
    _check_version(payload, "scenario")
    raw_assumptions = payload.get("assumptions")
    return Scenario(
        id=payload["id"],
        title=payload["title"],
        display_zone=payload["display_zone"],
        events=[event_from_dict(e) for e in payload.get("events", [])],
        windows=[_window_from_dict(w) for w in payload.get("windows", [])],
        assumptions=None if raw_assumptions is None else assumptions_from_dict(raw_assumptions),
    )


def save_scenario(scenario: Scenario, path: str | Path) -> None:
    Path(path).write_text(
        json.dumps(scenario_to_dict(scenario), indent=2) + "\n", encoding="utf-8"
    )


def load_scenario(path: str | Path) -> Scenario:
    return scenario_from_dict(json.loads(Path(path).read_text(encoding="utf-8")))


# --- Game weekends -------------------------------------------------------------


def _local_to_dict(time: LocalTime) -> dict[str, Any]:
    payload: dict[str, Any] = {"wall": time.wall, "zone": time.zone}
    if time.fold is not None:
        payload["fold"] = time.fold
    return payload


def _local_from_dict(payload: dict[str, Any]) -> LocalTime:
    return LocalTime(payload["wall"], payload["zone"], payload.get("fold"))


def _template_to_dict(template: KickoffBandTemplate) -> dict[str, Any]:
    return {
        "band": template.band,
        "game_type": template.game_type,
        "source": template.source,
        "entries": [
            {
                "id": entry.id,
                "kind": entry.kind,
                "title": entry.title,
                "start_minutes": _minutes(entry.start),
                "end_minutes": _minutes(entry.end),
                "start_at": entry.start_at,
                "end_at": entry.end_at,
                "movable": entry.movable,
            }
            for entry in template.entries
        ],
        "nights": [
            {
                "night": marker.night,
                "curfew_minutes": _minutes(marker.curfew),
                "wake_minutes": None if marker.wake is None else _minutes(marker.wake),
                "at": marker.at,
            }
            for marker in template.nights
        ],
    }


def _template_from_dict(payload: dict[str, Any]) -> KickoffBandTemplate:
    return KickoffBandTemplate(
        band=payload["band"],
        game_type=payload["game_type"],
        source=payload["source"],
        entries=tuple(
            TemplateEntry(
                id=entry["id"],
                kind=entry["kind"],
                title=entry["title"],
                start=timedelta(minutes=entry["start_minutes"]),
                end=timedelta(minutes=entry["end_minutes"]),
                start_at=entry.get("start_at", "home"),
                end_at=entry.get("end_at"),
                movable=entry.get("movable", False),
            )
            for entry in payload["entries"]
        ),
        nights=tuple(
            NightMarker(
                night=marker["night"],
                curfew=timedelta(minutes=marker["curfew_minutes"]),
                wake=None
                if marker.get("wake_minutes") is None
                else timedelta(minutes=marker["wake_minutes"]),
                at=marker.get("at", "home"),
            )
            for marker in payload["nights"]
        ),
    )


def weekend_to_dict(weekend: GameWeekend) -> dict[str, Any]:
    if weekend.kickoff_entered is None:
        raise ValueError(
            "this weekend does not record the kickoff as entered; build it with "
            "build_weekend so its inputs can be saved"
        )
    scenario = weekend.scenario
    return {
        "format_version": FORMAT_VERSION,
        "kind": WEEKEND_KIND,
        "id": scenario.id,
        "title": scenario.title,
        "kickoff": _local_to_dict(weekend.kickoff_entered),
        "home_zone": weekend.home_zone,
        "game_zone": weekend.game_zone,
        "assumptions": assumptions_to_dict(scenario.assumptions),
        "template": _template_to_dict(weekend.template),
        "offsets": {
            entry_id: {"start_minutes": _minutes(start), "end_minutes": _minutes(end)}
            for entry_id, (start, end) in weekend.offsets.items()
        },
        "window_overrides": {
            night: {
                "start": None if override.start is None else _local_to_dict(override.start),
                "end": None if override.end is None else _local_to_dict(override.end),
            }
            for night, override in weekend.window_overrides.items()
        },
        "extra_events": [event_to_dict(e) for e in weekend.extra_events],
    }


def weekend_from_dict(payload: dict[str, Any]) -> GameWeekend:
    kind = payload.get("kind")
    if kind != WEEKEND_KIND:
        raise ValueError(f"expected a {WEEKEND_KIND!r} file, got kind {kind!r}")
    _check_version(payload, WEEKEND_KIND)
    return build_weekend(
        _template_from_dict(payload["template"]),
        _local_from_dict(payload["kickoff"]),
        home_zone=payload["home_zone"],
        game_zone=payload["game_zone"],
        assumptions=assumptions_from_dict(payload["assumptions"]),
        id=payload["id"],
        title=payload["title"],
        offsets={
            entry_id: (
                timedelta(minutes=offset["start_minutes"]),
                timedelta(minutes=offset["end_minutes"]),
            )
            for entry_id, offset in payload.get("offsets", {}).items()
        },
        extra_events=[event_from_dict(e) for e in payload.get("extra_events", [])],
        window_overrides={
            night: WindowOverride(
                start=None if edge.get("start") is None else _local_from_dict(edge["start"]),
                end=None if edge.get("end") is None else _local_from_dict(edge["end"]),
            )
            for night, edge in payload.get("window_overrides", {}).items()
        },
    )


def save_weekend(weekend: GameWeekend, path: str | Path) -> None:
    Path(path).write_text(
        json.dumps(weekend_to_dict(weekend), indent=2) + "\n", encoding="utf-8"
    )


def load_weekend(path: str | Path) -> GameWeekend:
    return weekend_from_dict(json.loads(Path(path).read_text(encoding="utf-8")))
"""Sleep opportunity, and the comparison of two scenarios' opportunity.

This implements steps 1-8 of the spec's opportunity calculation. The two rules
worth restating, because every future change will be tempted to break them:

* Free time is not opportunity. Only an entered or proposed window counts.
* Opportunity is not sleep. Nothing here estimates sleep obtained, and the
  report names its own measure so a caller cannot mistake it for one.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import timedelta

from .schedule import Event, ProtectedSleepWindow, Scenario
from .temporal import Interval, merge, subtract, total_duration

__all__ = [
    "Comparison",
    "Conflict",
    "EventChange",
    "OpportunityReport",
    "compare",
    "evaluate",
    "event_changes",
]

MEASURE = "sleep opportunity"


def _hours(delta: timedelta) -> float:
    return round(delta.total_seconds() / 3600, 6)


@dataclass(frozen=True)
class Conflict:
    """An obligation that overlaps an entered sleep window.

    Reported, never resolved by deletion: the treatment stays on the schedule
    and the staff decides.
    """

    window_id: str
    event_id: str
    event_kind: str
    overlap: Interval

    def describe(self) -> str:
        return (
            f"{self.event_id} ({self.event_kind}) overlaps window {self.window_id} "
            f"for {_hours(self.overlap.duration)}h"
        )


@dataclass(frozen=True)
class OpportunityReport:
    """Opportunity within one reporting interval, with its provenance."""

    scenario: Scenario
    reporting_interval: Interval
    feasible: list[Interval]
    sources: list[ProtectedSleepWindow]
    conflicts: list[Conflict]
    target_hours: float
    target_source: str
    measure: str = MEASURE

    @property
    def hours(self) -> float:
        """Opportunity hours after union, so overlapping windows count once."""
        return _hours(total_duration(self.feasible))

    @property
    def shortfall_hours(self) -> float:
        return round(max(0.0, self.target_hours - self.hours), 6)


@dataclass(frozen=True)
class EventChange:
    """How one event differs between baseline and candidate.

    ``shift`` is the change in start and ``end_shift`` the change in end; an
    event that only ran longer has a zero ``shift``. ``movable`` is taken from
    the baseline, so a candidate cannot make a fixed event movable by
    redeclaring it.
    """

    event_id: str
    shift: timedelta
    end_shift: timedelta = timedelta()
    movable: bool = False


@dataclass(frozen=True)
class Comparison:
    baseline: OpportunityReport
    candidate: OpportunityReport
    changed_events: list[EventChange] = field(default_factory=list)
    measure: str = MEASURE

    @property
    def hours_delta(self) -> float:
        return round(self.candidate.hours - self.baseline.hours, 6)

    @property
    def resolved_conflicts(self) -> list[Conflict]:
        candidate_keys = {(c.window_id, c.event_id) for c in self.candidate.conflicts}
        return [c for c in self.baseline.conflicts if (c.window_id, c.event_id) not in candidate_keys]

    @property
    def fixed_event_changes(self) -> list[EventChange]:
        """Changes to events that were not levers. Flagged, never refused."""
        return [c for c in self.changed_events if not c.movable]

    @property
    def removed_events(self) -> list[str]:
        """Baseline events the candidate no longer has.

        Dropping an obligation raises opportunity, so it must be visible even
        though nothing "moved".
        """
        kept = {e.id for e in self.candidate.scenario.events}
        return [e.id for e in self.baseline.scenario.events if e.id not in kept]

    @property
    def added_events(self) -> list[str]:
        existing = {e.id for e in self.baseline.scenario.events}
        return [e.id for e in self.candidate.scenario.events if e.id not in existing]


def _blocked_intervals(scenario: Scenario) -> list[Interval]:
    """Mandatory obligations widened by the assumed transitions around them."""
    assumptions = scenario.assumptions
    pre_bed = assumptions.pre_bed if assumptions else timedelta()
    morning = assumptions.morning if assumptions else timedelta()
    blocked = []
    for event in scenario.obligations():
        blocked.append(
            Interval(event.interval.start - morning, event.interval.end + pre_bed)
        )
    return merge(blocked)


def evaluate(scenario: Scenario, reporting_interval: Interval) -> OpportunityReport:
    """Opportunity for ``scenario`` inside ``reporting_interval``.

    The reporting interval is supplied rather than derived so that an overnight
    total and a 24-hour total are two explicit calls, never one blended number.
    """
    if scenario.assumptions is None:
        raise ValueError("scenario needs an AssumptionSet before it can be evaluated")

    blocked = _blocked_intervals(scenario)

    conflicts: list[Conflict] = []
    feasible: list[Interval] = []
    sources: list[ProtectedSleepWindow] = []

    for window in scenario.windows:
        clipped = window.interval.intersect(reporting_interval)
        if clipped is None:
            continue
        sources.append(window)
        for event in scenario.obligations():
            overlap = clipped.intersect(event.interval)
            if overlap is not None:
                conflicts.append(
                    Conflict(window.id, event.id, event.kind, overlap)
                )
        feasible.extend(subtract(clipped, blocked))

    return OpportunityReport(
        scenario=scenario,
        reporting_interval=reporting_interval,
        feasible=merge(feasible),
        sources=sources,
        conflicts=conflicts,
        target_hours=scenario.assumptions.sleep_target_hours,
        target_source=scenario.assumptions.source,
    )


def _shifts(original: Event, event: Event) -> tuple[timedelta, timedelta]:
    # Events placed from kickoff are compared by offset. Otherwise a kickoff
    # change would report every anchored event as moved, burying the one
    # change staff actually made.
    if original.anchor is not None and event.anchor is not None:
        return event.anchor.start - original.anchor.start, event.anchor.end - original.anchor.end
    return event.interval.start - original.interval.start, event.interval.end - original.interval.end


def event_changes(baseline: Scenario, candidate: Scenario) -> list[EventChange]:
    """Events present in both scenarios whose start or end differs."""
    before = {e.id: e for e in baseline.events}
    changes = []
    for event in candidate.events:
        original = before.get(event.id)
        if original is None:
            continue
        start_shift, end_shift = _shifts(original, event)
        if start_shift or end_shift:
            changes.append(EventChange(event.id, start_shift, end_shift, original.movable))
    return changes


def compare(baseline: OpportunityReport, candidate: OpportunityReport) -> Comparison:
    """Difference between two scenarios evaluated on common ground.

    Refuses mismatched assumptions or reporting intervals: a difference in
    opportunity is only interpretable when everything else was held equal.
    """
    if baseline.scenario.assumptions != candidate.scenario.assumptions:
        raise ValueError(
            "cannot compare scenarios built on a different assumption set; "
            f"baseline target {baseline.target_hours}h from {baseline.target_source!r}, "
            f"candidate target {candidate.target_hours}h from {candidate.target_source!r}"
        )
    if baseline.reporting_interval != candidate.reporting_interval:
        raise ValueError("cannot compare reports covering different reporting intervals")

    return Comparison(
        baseline=baseline,
        candidate=candidate,
        changed_events=event_changes(baseline.scenario, candidate.scenario),
    )

"""The protected sleep window rule, per night of a game weekend.

start = the later of curfew and (last obligation end + wind-down buffer)
end   = the wake-up call; Saturday night ends at an assumed wake time instead,
        because Sunday is a player off day and nothing else bounds it.

Synthetic weekend (see tests/synthetic.py): Friday curfew 23:00 MT, wake-up
06:00 MT, meetings 21:00-22:00 MT, 30-minute wind-down buffer.
"""

from dataclasses import replace
from datetime import timedelta

import pytest

from sleep_scheduler.opportunity import evaluate
from sleep_scheduler.temporal import Interval, LocalTime
from sleep_scheduler.weekend import WindowOverride, build_weekend
from synthetic import ASSUMPTIONS, CHICAGO, DENVER, EARLY_KICKOFF, H, road_early_template

WITH_WAKE = replace(ASSUMPTIONS, saturday_night_wake="09:00")
NO_WAKE = replace(ASSUMPTIONS, saturday_night_wake=None)
M = timedelta(minutes=1)


def build(assumptions=WITH_WAKE, **kwargs):
    return build_weekend(
        road_early_template(),
        EARLY_KICKOFF,
        home_zone=CHICAGO,
        game_zone=DENVER,
        assumptions=assumptions,
        id="baseline",
        title="Early road kickoff (synthetic)",
        **kwargs,
    )


def night(weekend, name):
    return next(n for n in weekend.nights if n.night == name)


def test_friday_window_runs_from_curfew_to_wake_up_when_curfew_binds():
    window = night(build(), "fri").window
    assert window.interval.wall("start") == "2026-10-02T23:00"
    assert window.interval.wall("end") == "2026-10-03T06:00"
    assert (window.start_source, window.end_source) == ("curfew", "wake-up call")


def test_away_night_is_a_hotel_window_and_home_nights_are_home_windows():
    weekend = build()
    assert [night(weekend, n).window.kind for n in ("thu", "fri", "sat")] == ["home", "hotel", "home"]


def test_moving_friday_meetings_earlier_leaves_the_window_unchanged_while_curfew_binds():
    base = night(build(), "fri").window.interval
    moved = night(build(offsets={"fri-meeting": (-16 * H, -15 * H)}), "fri").window.interval
    assert moved == base


def test_meetings_ending_after_curfew_minus_buffer_shorten_the_night_by_the_overrun():
    # Meetings now end 22:50 MT; plus 30 minutes is 23:20, 20 minutes past curfew.
    base = night(build(), "fri").window
    late = night(build(offsets={"fri-meeting": (-13 * H, -12 * H - 10 * M)}), "fri").window
    assert late.interval.wall("start") == "2026-10-02T23:20"
    assert late.start_source == "obligation:fri-meeting"
    assert base.interval.duration - late.interval.duration == 20 * M


def test_the_wind_down_buffer_comes_from_the_assumption_set():

    return weekend_from_dict(json.loads(json.dumps(weekend_to_dict(weekend))))

"""Instants, half-open intervals, and the interval algebra the spec requires.

The spec's temporal rules are the reason this module exists rather than bare
``datetime`` use: a local wall time plus an IANA zone is not always a single
instant, and the two cases where it is not must be refused rather than guessed.
Elapsed duration is always computed in UTC; zones are retained for rendering.
"""

from __future__ import annotations

from dataclasses import dataclass, field
from datetime import datetime, timedelta, timezone
from typing import Iterable, Sequence
from zoneinfo import ZoneInfo

__all__ = [
    "AmbiguousLocalTime",
    "Interval",
    "InvalidInterval",
    "LocalTime",
    "NonexistentLocalTime",
    "TemporalError",
    "merge",
    "subtract",
    "total_duration",
]


class TemporalError(ValueError):
    """Base class for a local time or interval that cannot be trusted."""


class NonexistentLocalTime(TemporalError):
    """The wall time never occurs in that zone (spring-forward gap)."""


class AmbiguousLocalTime(TemporalError):
    """The wall time occurs twice in that zone and no fold was supplied."""


class InvalidInterval(TemporalError):
    """An interval whose end does not strictly follow its start."""


@dataclass(frozen=True)
class LocalTime:
    """A wall time entered by staff, kept alongside the zone it was entered in.

    ``fold`` is only meaningful for a repeated wall time: 0 selects the first
    occurrence (before the clocks go back), 1 the second. Supplying it for an
    unambiguous time is harmless and ignored.
    """

    wall: str
    zone: str
    fold: int | None = None

    def __post_init__(self) -> None:
        naive = datetime.fromisoformat(self.wall)
        if naive.tzinfo is not None:
            raise TemporalError(
                f"{self.wall!r} already carries an offset; pass a bare wall time "
                "and an IANA zone so the entered local context is preserved"
            )
        tz = ZoneInfo(self.zone)

        # A gap is detected by round-tripping: for a time that never occurred,
        # zoneinfo silently shifts it, so the return trip lands somewhere else.
        probe = naive.replace(tzinfo=tz)
        if probe.astimezone(timezone.utc).astimezone(tz).replace(tzinfo=None) != naive:
            raise NonexistentLocalTime(
                f"{self.wall} does not occur in {self.zone}; clocks skip that hour"
            )

        # A repeated time has two distinct offsets. zoneinfo would default to
        # fold=0 and quietly pick one, which the spec forbids.
        if probe.utcoffset() != naive.replace(tzinfo=tz, fold=1).utcoffset():
            if self.fold is None:
                raise AmbiguousLocalTime(
                    f"{self.wall} occurs twice in {self.zone}; supply fold=0 or fold=1"
                )

        object.__setattr__(self, "_naive", naive)
        object.__setattr__(self, "_tz", tz)

    @property
    def instant(self) -> datetime:
        """The single UTC instant this wall time and zone denote."""
        naive: datetime = getattr(self, "_naive")
        tz: ZoneInfo = getattr(self, "_tz")
        return naive.replace(tzinfo=tz, fold=self.fold or 0).astimezone(timezone.utc)

    def render(self, zone: str) -> str:
        """The same instant as wall time in another zone, for display only."""
        return self.instant.astimezone(ZoneInfo(zone)).isoformat()


@dataclass(frozen=True, order=True)
class Interval:
    """A half-open ``[start, end)`` span of UTC instants.

    Half-open is what lets a 22:00 event and a 22:00 window sit adjacent
    without being reported as a conflict.
    """

    start: datetime
    end: datetime
    # Provenance for rendering and export. Excluded from comparison so an
    # interval derived by merge/subtract still equals an equivalent entered one.
    start_zone: str | None = field(default=None, compare=False)
    end_zone: str | None = field(default=None, compare=False)
    start_fold: int | None = field(default=None, compare=False)
    end_fold: int | None = field(default=None, compare=False)

    def __post_init__(self) -> None:
        for name, value in (("start", self.start), ("end", self.end)):
            if value.tzinfo is None:
                raise InvalidInterval(f"{name} must be an aware instant, got {value!r}")
        object.__setattr__(self, "start", self.start.astimezone(timezone.utc))
        object.__setattr__(self, "end", self.end.astimezone(timezone.utc))
