#!/usr/bin/env bash
# Install the TypeSafe API key once, globally, on macOS.
#
# The key goes into the macOS Keychain (service: typesafe-api-key). A line added to
# ~/.zshenv exports it as TYPESAFE_API_KEY for every zsh, including the
# non-interactive shells Claude Code spawns. That line holds no secret, so it is
# safe even if your dotfiles are in a git repo.
#
# Run this in Terminal.app, NOT through Claude Code. The prompt needs a real
# terminal, and anything typed into a Claude session is written to a transcript
# that is tracked in the dotclaude repo.
#
# Usage:  ./setup_global_key.sh          install or update the key
#         ./setup_global_key.sh --check  report status, change nothing

set -euo pipefail

SERVICE="typesafe-api-key"
ZSHENV="$HOME/.zshenv"
MARKER="# >>> typesafe api key >>>"
END_MARKER="# <<< typesafe api key <<<"

if [[ "$(uname -s)" != "Darwin" ]]; then
  echo "This installer is macOS only (it uses the Keychain). See docs/STATUS.md for Windows." >&2
  exit 1
fi

keychain_has_key() {
  security find-generic-password -a "$USER" -s "$SERVICE" -w >/dev/null 2>&1
}

zshenv_has_block() {
  [[ -f "$ZSHENV" ]] && grep -qF "$MARKER" "$ZSHENV"
}

if [[ "${1:-}" == "--check" ]]; then
  keychain_has_key && echo "keychain:  key present (service $SERVICE)" || echo "keychain:  MISSING"
  zshenv_has_block && echo "zshenv:    export block present" || echo "zshenv:    MISSING"
  echo "this shell: TYPESAFE_API_KEY is ${TYPESAFE_API_KEY:+set}${TYPESAFE_API_KEY:-NOT SET}"
  exit 0
fi

# --- 1. get the key, without it ever reaching a screen, a log, or shell history ---
if [[ ! -t 0 ]]; then
  echo "ERROR: no terminal on stdin." >&2
  echo "Open Terminal.app and run this script there. Do not run it through Claude Code:" >&2
  echo "the prompt cannot read input, and the key would land in a tracked transcript." >&2
  exit 1

if keychain_has_key; then
  echo "A key is already in the Keychain under service '$SERVICE'."
  printf "Replace it? [y/N] "

printf "Paste your TypeSafe API key (input is hidden, from https://console.typesafe.ai/): "

"""Prove the global TypeSafe API key resolves and the System One API answers.

There is deliberately no python-dotenv here. The key is meant to be available
globally from the environment (macOS Keychain -> ~/.zshenv -> every zsh), so
reading anything other than os.environ would hide a broken setup.

Run:  python smoke_test.py
"""

import os
import sys

REQUIRED_VAR = "TYPESAFE_API_KEY"


def require_key() -> None:
    if os.environ.get(REQUIRED_VAR):
        return
    print(f"{REQUIRED_VAR} is not set in this environment.\n", file=sys.stderr)
    print("Fix it in this order:", file=sys.stderr)
    print("  1. Run ./setup_global_key.sh in Terminal.app (not through Claude Code).", file=sys.stderr)
    print("  2. Restart Claude Code. A running session keeps the environment it", file=sys.stderr)
    print("     started with, so it will not see the new variable until restarted.", file=sys.stderr)
    print("  3. Check status any time with ./setup_global_key.sh --check", file=sys.stderr)
    sys.exit(1)


def main() -> int:
    require_key()

    try:
        from typesafe_sdk import Noul, TypeSafeClient
    except ModuleNotFoundError:
        print("typesafe-sdk is not installed. Run:  pip install typesafe-sdk", file=sys.stderr)
        return 1

    # The client picks the key up from TYPESAFE_API_KEY on its own.
    with TypeSafeClient() as client:
        response = client.system_one(
            state={"document": "I was charged twice. Please fix this ASAP."},
            questions={"billing": Noul(instructions="Is this ticket about billing?")},
        )

    probability = response.nouls["billing"].noul
    print(f"question: is this ticket about billing?")
    print(f"answer:   {probability}")
    print()
    print("The key resolved from the environment and the API answered. Setup is working.")
    return 0


if __name__ == "__main__":
    sys.exit(main())
EOF
