+"""The live season, one week at a time. Only weeks whose completed games lack plays or drives are pulled, so a run
+that stops partway resumes at the first week still missing, and a run with nothing new costs only the schedule calls.
+
+The live season is built with the Phase 1 builders scoped to that one season; thresholds, cohorts and the wins model
+read config.seasons() only, so the live season never enters the reference."""
+from __future__ import annotations
+
+from datetime import datetime
+
+import duckdb
+
+from cfb_benchmarks.adjust import adjust_team_game_metrics
+from cfb_benchmarks.config import Config
+from cfb_benchmarks.derive_plays import derive_plays
+from cfb_benchmarks.ingest_drives import _store_drives
+from cfb_benchmarks.ingest_fbs_teams import school_to_id
+from cfb_benchmarks.ingest_plays import EmptyTeamMapError, _store_plays, check_week_rows
+from cfb_benchmarks.metrics import build_team_game_metrics
+from cfb_benchmarks.season_metrics import build_team_season_metrics
+
+SCHEDULE_CALLS = 3          # /teams/fbs plus /games for regular and postseason
+
+
+def missing_weeks(con: duckdb.DuckDBPyConnection, season: int) -> list[tuple[str, int]]:
+    """(season_type, week) with a completed game that has an FBS side but no stored plays or no stored drives."""
+    rows = con.execute(
+        """with done as (
+               select distinct g.season_type, g.week from games g
+               where g.season = ? and g.home_points is not null and g.away_points is not null
+                 and (g.home_id in (select cfbd_id from fbs_teams f where f.season = g.season)
+                      or g.away_id in (select cfbd_id from fbs_teams f where f.season = g.season))),
+           p as (select distinct season_type, week from plays where season = ?),
+           d as (select distinct season_type, week from drives where season = ?)
+           select done.season_type, done.week from done
+           left join p using (season_type, week) left join d using (season_type, week)
+           where p.week is null or d.week is null""",
+        [season, season, season],
+    ).fetchall()
+    return sorted(((st, int(w)) for st, w in rows), key=lambda r: (r[0] != "regular", r[1]))
+
+
+def pull_week(con: duckdb.DuckDBPyConnection, client, config: Config, season: int, season_type: str, week: int,
+              pulled_at: datetime) -> tuple[int, int]:
+    """Plays then drives for one week, each checked before it is stored. A refused endpoint stores nothing."""
+    ids = school_to_id(con, season)
+    if not ids:
+        raise EmptyTeamMapError(season)
+    plays = client.plays(season, week, season_type)
+    check_week_rows(con, config, season, week, season_type, plays)
+    n_plays = _store_plays(con, season, week, season_type, plays, ids, pulled_at)
+    drives = client.drives(season, week, season_type)
+    check_week_rows(con, config, season, week, season_type, drives)
+    n_drives = _store_drives(con, season, week, season_type, drives, ids, pulled_at)
+    return n_plays, n_drives
+
+
+def estimate_live_calls(n_weeks: int) -> int:
+    return SCHEDULE_CALLS + 2 * n_weeks
+
+
+def build_live_season(con: duckdb.DuckDBPyConnection, config: Config, built_at: datetime, season: int) -> dict[str, int]:
+    seasons = [season]
+    derive_plays(con, config, seasons)
+    build_team_game_metrics(con, config, built_at, seasons)
+    adjust_team_game_metrics(con, config, built_at, seasons)
+    build_team_season_metrics(con, config, built_at, seasons, from_cohorts=False)
+    return {t: con.execute(f"select count(*) from {t} where season = ?", [season]).fetchone()[0]
+            for t in ("team_game_metrics", "team_season_metrics")}

+def monitor_strip(row: dict, width: int = 300, height: int = 26) -> str:
+    """The reference band (average cut to elite cut), a tick at the playoff-team median, and this game's dot.
+    Better is always to the right. The dot carries --out only when the value is outside the band on the bad side."""
+    pts = [row.get(k) for k in ("cut_average", "cut_elite", "cfp_median", "value")]
+    vals = [v for v in pts if v is not None]
+    if not vals or row.get("cut_average") is None or row.get("cut_elite") is None:
+        return (f'<svg class="strip" data-series="{row["name"]}" width="{width}" height="{height}" '
+                f'viewBox="0 0 {width} {height}" role="img" aria-label="no reference range"></svg>')
+    lo, hi = min(vals), max(vals)
+    span = (hi - lo) or 1.0
+    lo, hi = lo - 0.1 * span, hi + 0.1 * span
+    span = hi - lo
+    pad = 8
+
+    def x(v: float) -> float:
+        frac = (v - lo) / span if row["direction"] == "higher" else (hi - v) / span
+        return round(pad + frac * (width - 2 * pad), 1)
+
+    xa, xe = x(row["cut_average"]), x(row["cut_elite"])
+    mid = height / 2
+    parts = [f'<svg class="strip" data-series="{row["name"]}" width="{width}" height="{height}" viewBox="0 0 {width} {height}" role="img" '
+             f'aria-label="{row["label"]} against the reference range">',
+             f'<line x1="{pad}" y1="{mid}" x2="{width - pad}" y2="{mid}" class="axis"/>',
+             f'<rect data-label="band" x="{min(xa, xe)}" y="{mid - 6}" width="{abs(xe - xa)}" height="12" class="band"/>',
+             f'<line data-label="cut_average" x1="{xa}" y1="{mid - 6}" x2="{xa}" y2="{mid + 6}" class="cut"/>',
+             f'<line data-label="cut_elite" x1="{xe}" y1="{mid - 6}" x2="{xe}" y2="{mid + 6}" class="cut"/>']
+    if row.get("cfp_median") is not None:
+        xm = x(row["cfp_median"])
+        parts.append(f'<line data-label="cfp_median" x1="{xm}" y1="{mid - 10}" x2="{xm}" y2="{mid + 10}" class="cfp"/>')
+    if row.get("value") is not None:
+        cls = "dot out" if row.get("out_of_range") else "dot"
+        parts.append(f'<circle data-label="value" cx="{x(row["value"])}" cy="{mid}" r="4.5" class="{cls}"/>')
+    parts.append("</svg>")
+    return "".join(parts)

+"""Parquet snapshot of every table, the way machines exchange this DB (the DB itself never travels).
+
+Files are written to a local temp folder, then copied into the target folder, then the manifest is written last, so
+a reader never sees a manifest that promises files not yet there. A snapshot never shrinks without --force: if the
+published manifest records more rows for any table than the DB holds, the DB is probably an older copy."""
+from __future__ import annotations
+
+import json
+import shutil
+import tempfile
+from datetime import datetime
+from pathlib import Path
+
+import duckdb
+
+MANIFEST = "manifest.json"
+
+
+class SnapshotRefused(Exception):
+    def __init__(self, shrinking: dict[str, tuple[int, int]]):
+        self.shrinking = shrinking
+        detail = ", ".join(f"{t} {pub} published, {db} in the DB" for t, (pub, db) in sorted(shrinking.items()))
+        super().__init__(f"snapshot would shrink: {detail}. Pass --force-snapshot if the DB is right.")
+
+
+def export_snapshot(con: duckdb.DuckDBPyConnection, out_dir: Path, exported_at: datetime, force: bool = False) -> dict[str, int]:
+    out_dir = Path(out_dir)
+    tables = sorted(r[0] for r in con.execute("show tables").fetchall())
+    rows = {t: con.execute(f'select count(*) from "{t}"').fetchone()[0] for t in tables}
+    manifest = out_dir / MANIFEST
+    if manifest.exists() and not force:
+        published = json.loads(manifest.read_text()).get("rows", {})
+        shrinking = {t: (int(published[t]), rows.get(t, 0)) for t in published if int(published[t]) > rows.get(t, 0)}
+        if shrinking:
+            raise SnapshotRefused(shrinking)
+    out_dir.mkdir(parents=True, exist_ok=True)
+    with tempfile.TemporaryDirectory() as tmp:
+        for t in tables:
+            local = Path(tmp) / f"{t}.parquet"
+            con.execute(f"copy \"{t}\" to '{local}' (format parquet)")
+            shutil.copy2(local, out_dir / local.name)
+    manifest.write_text(json.dumps({"exported_at": exported_at.isoformat(), "rows": rows}, indent=2))
+    return rows
