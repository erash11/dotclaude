======== src/coachreport/payload.py ========
     1	"""Payload assembly. The payload is the frozen interface to the templates.
     2	
     3	Every weekly variable a sheet prints is derived once, here, into `meta`. No
     4	opponent name, date, week number or season label may be typed into markup — that
     5	is the failure that killed the 2024 report. The same rule covers computed prose:
     6	a claim like "the widest gap of the season" is generated from the rows rendered
     7	beside it, never written down.
     8	"""
     9	from collections import defaultdict
    10	from datetime import datetime
    11	
    12	from . import config, flags
    13	from .metrics import band_rows
    14	from .resolve import ResolvedWeek

    17	def build_meta(snap: dict, rw: ResolvedWeek) -> dict:
    18	    game = snap["game"]
    19	    dt = datetime.fromisoformat(str(game["date"]).replace("Z", "+00:00"))
    20	    home = game["home"] == config.TEAM
    21	
    22	    meta = {
    23	        "team": config.TEAM,
    24	        "opponent": rw.opponent,
    25	        "next_opponent": rw.next_opponent,
    26	        "season": rw.season,
    27	        "week": rw.week,
    28	        "date": dt.strftime("%B %-d, %Y"),
    29	        "date_short": dt.strftime("%d %b %Y"),
    30	        "venue": game["venue"],
    31	        "team_score": game["home_points"] if home else game["away_points"],
    32	        "opp_score": game["away_points"] if home else game["home_points"],
    33	        "excitement": (round(game["excitement"], 2)
    34	                       if game.get("excitement") is not None else None),
    35	        "home": home,
    36	    }
    37	    meta["result"] = "W" if meta["team_score"] > meta["opp_score"] else "L"
    38	
    39	    if rw.next_opponent:
    40	        vs = "vs" if rw.next_home else "at"
    41	        meta["next_week"] = rw.next_week
    42	        meta["next_home"] = rw.next_home
    43	        meta["next_game"] = f"Week {rw.next_week} · {vs} {rw.next_opponent}"
    44	        meta["next_short"] = f"Wk {rw.next_week} {vs} {rw.next_opponent}"
    45	    else:
    46	        meta["next_week"] = meta["next_home"] = None
    47	        meta["next_game"] = meta["next_short"] = "No further games"
    48	
    49	    prior = list(range(1, rw.week))
    50	    meta["week_label"] = f"week {rw.week}"
    51	    meta["season_range"] = f"weeks {prior[0]}–{rw.week}" if prior else f"week {rw.week}"
    52	    if len(prior) > 1:
    53	        meta["baseline_range"] = f"weeks {prior[0]}–{prior[-1]}"
    54	        meta["baseline_range_prose"] = f"weeks {prior[0]} to {prior[-1]}"
    55	    elif prior:
    56	        meta["baseline_range"] = meta["baseline_range_prose"] = f"week {prior[0]}"
    57	    else:
    58	        meta["baseline_range"] = meta["baseline_range_prose"] = "no prior games"
    59	    return meta

    62	def _tempo(drives: list[dict]) -> dict:
    63	    plays = sum(d["plays"] for d in drives)
    64	    secs = sum(d["elapsed"]["minutes"] * 60 + d["elapsed"]["seconds"] for d in drives)
    65	    yards = sum(d["yards"] for d in drives)
    66	    n = len(drives)
    67	    return {"drives": n, "plays": plays, "yards": yards, "sec": secs,
    68	            "plays_per_drive": round(plays / n, 2) if n else 0,
    69	            "sec_per_play": round(secs / plays, 1) if plays else 0,
    70	            "yards_per_drive": round(yards / n, 1) if n else 0,
    71	            "scoring_drives": sum(1 for d in drives if d["scoring"]),
    72	            "three_and_outs": sum(1 for d in drives
    73	                                  if d["plays"] <= 3 and not d["scoring"])}

    76	def build_payloads(snap: dict, rw: ResolvedWeek) -> tuple[dict, dict]:
    77	    team, week = config.TEAM, rw.week
    78	    meta = build_meta(snap, rw)
    79	    dist = snap["p4"]["dist"]
    80	
    81	    # ---------------------------------------------------------- shared spine
    82	    # One definition of the week's snap count, computed once and copied. Two
    83	    # editions computing this separately is how they drifted apart the first time.
    84	    bx = next((g for g in snap["adv_games_baylor"] if g["week"] == week), None)
    85	    box_snaps = ({"off": bx["offense"]["plays"], "def": bx["defense"]["plays"],
    86	                  "off_drives": bx["offense"]["drives"],
    87	                  "def_drives": bx["defense"]["drives"]} if bx else None)
    88	
    89	    drives = snap["drives"]
    90	    tempo = {"baylor_off": _tempo([d for d in drives if d["offense"] == team]),
    91	             "baylor_def": _tempo([d for d in drives if d["offense"] != team])}
    92	    drive_log = [{"num": d["driveNumber"],
    93	                  "off": "BU" if d["offense"] == team else "OPP",
    94	                  "plays": d["plays"], "yards": d["yards"], "result": d["driveResult"],
    95	                  "start_ytg": d["startYardsToGoal"],
    96	                  "elapsed": d["elapsed"]["minutes"] * 60 + d["elapsed"]["seconds"],
    97	                  "period": d["startPeriod"]}
    98	                 for d in sorted(drives, key=lambda z: z["driveNumber"])]
    99	
   100	    # per-player game snaps: max across grade types is the true snap count
   101	    snaps, grade_map = {}, defaultdict(dict)
   102	    for g in snap["pff_grades"]:
   103	        k = (g["team_id"], g["name"], g["position"])
   104	        snaps[k] = max(snaps.get(k, 0), g["snaps"] or 0)
   105	        grade_map[k][g["grade_type"]] = g["grade"]
   106	
   107	    tid = snap["meta"]["pff_team_id"]
   108	    bay_rows = []
   109	    for (t, name, pos), gm in grade_map.items():
   110	        if t != tid:
   111	            continue
   112	        row = {"name": name, "position": pos, "group": config.group_of(pos),
   113	               "side": config.side_of(pos), "snaps": snaps[(t, name, pos)]}
   114	        row.update({k: round(v, 1) for k, v in gm.items() if v is not None})
   115	        bay_rows.append(row)
   116	
   117	    # ------------------------------------------------------ coaches edition
   118	    if not snap["adv_game"]:
   119	        raise ValueError(
   120	            f"no advanced box score for game {rw.game_id} "
   121	            f"({config.TEAM} week {rw.week}, {rw.season}) — CFBD has not posted it yet")
   122	
   123	    wp = snap["winprob_raw"]
   124	    winprob = [{"p": w["playNumber"], "wp": round(w["homeWinProbability"], 4),
   125	                "hs": w["homeScore"], "as": w["awayScore"]} for w in wp]
   126	    events = []
   127	    for prev, cur in zip(wp, wp[1:]):
   128	        if cur["homeScore"] != prev["homeScore"] or cur["awayScore"] != prev["awayScore"]:
   129	            events.append({"p": cur["playNumber"], "hs": cur["homeScore"],
   130	                           "as": cur["awayScore"],
   131	                           "wp": round(cur["homeWinProbability"], 3),
   132	                           "swing": round(cur["homeWinProbability"]
   133	                                          - prev["homeWinProbability"], 3),
   134	                           "text": (cur.get("playText") or "")[:96]})
   135	
   136	    A = {
   137	        "meta": meta,
   138	        "p4_n": snap["p4"]["n_teams"],
   139	        "bands": band_rows(snap["baylor_season_adv"], dist),
   140	        "next_bands": band_rows(snap["next_opp_season_adv"], dist),
   141	        "winprob": winprob,
   142	        "score_events": events,
   143	        "box": {"offense": snap["adv_game"][0]["offense"],
   144	                "defense": snap["adv_game"][0]["defense"]},
   145	        "ppa_game": next((g for g in snap["ppa_games"]
   146	                          if g["gameId"] == rw.game_id), None),
   147	        "ppa_season": snap["ppa_team_season"][0] if snap["ppa_team_season"] else {},
   148	        "epa_by_week": [{"week": g["week"], "opponent": g["opponent"],
   149	                         "off": g["offense"]["overall"], "def": g["defense"]["overall"]}
   150	                        for g in sorted(snap["ppa_games"], key=lambda z: z["week"])],
   151	        "drive_log": drive_log,
   152	        "tempo": tempo,
   153	        "box_snaps": box_snaps,
   154	        "pff_offense": sorted(
   155	            [{k: r[k] for k in ("name", "position", "group", "snaps", "offense")}
   156	             for r in bay_rows if r.get("offense") is not None and r["snaps"] >= 15],
   157	            key=lambda z: (-z["offense"], z["name"]))[:11],
   158	        "pff_defense": sorted(
   159	            [{k: r[k] for k in ("name", "position", "group", "snaps", "defense")}
   160	             for r in bay_rows if r.get("defense") is not None and r["snaps"] >= 15],
   161	            key=lambda z: (-z["defense"], z["name"]))[:11],
   162	    }
   163	
   164	    # -------------------------------------------------- performance edition
   165	    q_off, q_def = defaultdict(int), defaultdict(int)
   166	    for p in snap["plays"]:
   167	        (q_off if p["offense"] == team else q_def)[p["period"]] += 1
   168	
   169	    def tempo_rows(adv):
   170	        return [{"week": g["week"], "opponent": g["opponent"],
   171	                 "off_plays": g["offense"]["plays"], "def_plays": g["defense"]["plays"],
   172	                 "off_drives": g["offense"]["drives"],
   173	                 "def_drives": g["defense"]["drives"]}
   174	                for g in sorted(adv, key=lambda z: z["week"]) if g["week"] <= week]
   175	
   176	    grp = defaultdict(lambda: {"snaps": 0, "n": 0})
   177	    for r in bay_rows:
   178	        grp[r["group"]]["snaps"] += r["snaps"]
   179	        grp[r["group"]]["n"] += 1
   180	
   181	    series, pos = defaultdict(dict), {}
   182	    for r in snap["pff_season_snaps"]:
   183	        series[r["name"]][int(r["week"])] = int(r["snaps"] or 0)
   184	        pos[r["name"]] = r["position"]
   185	    weeks = sorted({int(r["week"]) for r in snap["pff_season_snaps"]})
   186	
   187	    season_players = []
   188	    for name, by_week in series.items():
   189	        prior = [v for w, v in by_week.items() if w < week and v > 0]
   190	        this = by_week.get(week, 0)
   191	        band = flags.band_for(prior)
   192	        base = sum(prior) / len(prior) if prior else 0
   193	        season_players.append({
   194	            "name": name, "position": pos[name],
   195	            "group": config.group_of(pos[name]), "side": config.side_of(pos[name]),
   196	            "by_week": {str(w): by_week.get(w, 0) for w in weeks},
   197	            "total": sum(by_week.values()),
   198	            "games": len([v for v in by_week.values() if v > 0]),
   199	            "this_week": this, "baseline": round(base, 1), "band": band,
   200	            "delta_pct": round(100 * (this - base) / base, 1) if base else None,
   201	            "dev": round(this - band["med"], 1) if band else None,
   202	            "flag": flags.classify(this, band),
   203	        })
   204	
   205	    B = {
   206	        "meta": meta,
   207	        "tempo": tempo,
   208	        "snaps_by_quarter": [{"q": q, "off": q_off.get(q, 0), "def": q_def.get(q, 0)}
   209	                             for q in sorted(set(q_off) | set(q_def))],
   210	        "drive_log": drive_log,
   211	        "season_tempo": tempo_rows(snap["adv_games_baylor"]),
   212	        "next_opp_tempo": tempo_rows(snap["next_opp_adv_games"]),
   213	        "box_snaps": box_snaps,          # copied from A, never recomputed
   214	        "group_totals": [{"group": g, "side": config.SIDE.get(g, "st"),
   215	                          "snaps": v["snaps"], "players": v["n"],
   216	                          "avg": round(v["snaps"] / v["n"], 1)}
   217	                         for g, v in sorted(grp.items(),
   218	                                            key=lambda kv: (-kv[1]["snaps"], kv[0]))],
   219	        "weeks": weeks,
   220	        "flag_rule": flags.RULE,
   221	        "season_players": sorted(season_players,
   222	                                 key=lambda z: (-z["this_week"], z["name"])),
   223	    }
   224	    return A, B
======== src/coachreport/resolve.py ========
     1	"""Derives every weekly variable from the schedule.
     2	
     3	The predecessor report died of hand-editing: the opponent, the week and the next
     4	game were typed in every Monday. Nothing in this module accepts a typed value
     5	except an explicit `--week` override for rebuilding an old week.
     6	"""
     7	from dataclasses import dataclass
     8	from datetime import datetime
     9	
    10	from . import config
    11	
    12	
    13	class NoCompletedGame(RuntimeError):
    14	    """No finished regular-season game to report on — a bye, or the season hasn't started."""
    15	
    16	
    17	@dataclass(frozen=True)
    18	class ResolvedWeek:
    19	    season: int
    20	    week: int
    21	    game_id: int
    22	    opponent: str
    23	    home: bool
    24	    next_opponent: str | None
    25	    next_week: int | None
    26	    next_home: bool | None
    27	
    28	
    29	def current_season(now: datetime | None = None) -> int:
    30	    now = now or datetime.now()
    31	    return now.year if now.month >= config.SEASON_START_MONTH else now.year - 1
    32	
    33	
    34	def _regular(games: list[dict]) -> list[dict]:
    35	    return [g for g in games if g.get("seasonType", "regular") == "regular"]
    36	
    37	
    38	def _played(game: dict) -> bool:
    39	    return game.get("homePoints") is not None and game.get("awayPoints") is not None
    40	
    41	
    42	def resolve_week(games: list[dict], team: str, season: int,
    43	                 week: int | None = None) -> ResolvedWeek:
    44	    reg = _regular(games)
    45	    done = [g for g in reg if _played(g)]
    46	
    47	    if week is None:
    48	        if not done:
    49	            raise NoCompletedGame(
    50	                f"{team} has no completed regular-season game in {season}. "
    51	                "Nothing to report — this is a bye week or the season has not started.")
    52	        game = max(done, key=lambda g: g["week"])
    53	    else:
    54	        game = next((g for g in done if g["week"] == week), None)
    55	        if game is None:
    56	            raise NoCompletedGame(
    57	                f"{team} has no completed regular-season game in week {week} of {season}.")
    58	
    59	    home = game["homeTeam"] == team
    60	    opponent = game["awayTeam"] if home else game["homeTeam"]
    61	
    62	    later = [g for g in reg if g["week"] > game["week"]]
    63	    nxt = min(later, key=lambda g: g["week"]) if later else None
    64	
    65	    return ResolvedWeek(
    66	        season=season,
    67	        week=game["week"],
    68	        game_id=game["id"],
    69	        opponent=opponent,
    70	        home=home,
    71	        next_opponent=(nxt["awayTeam"] if nxt["homeTeam"] == team else nxt["homeTeam"])
    72	        if nxt else None,
    73	        next_week=nxt["week"] if nxt else None,
    74	        next_home=(nxt["homeTeam"] == team) if nxt else None,
    75	    )

     8	P4_METRICS = {
     9	    "off_success":    ("offense", "successRate"),

    11	    "off_ppa":        ("offense", "ppa"),
    12	    "off_rush_ppa":   ("offense", "rushingPlays", "ppa"),
    13	    "off_pass_ppa":   ("offense", "passingPlays", "ppa"),
    14	    "off_line_yards": ("offense", "lineYards"),
    15	    "off_stuff":      ("offense", "stuffRate"),
    16	    "off_havoc":      ("offense", "havoc", "total"),
    17	    "def_success":    ("defense", "successRate"),
    18	    "def_explosive":  ("defense", "explosiveness"),
    19	    "def_ppa":        ("defense", "ppa"),
    20	    "def_havoc":      ("defense", "havoc", "total"),
    21	    "def_line_yards": ("defense", "lineYards"),
    22	    "def_stuff":      ("defense", "stuffRate"),
    23	}

    26	def dig(d, *keys):
    27	    for k in keys:
    28	        if not isinstance(d, dict):
    29	            return None
    30	        d = d.get(k)
    31	    return d
    32	
    33	
    34	def p4_distributions(all_season_adv: list[dict], conf_by_team: dict) -> dict:
    35	    rows = [r for r in all_season_adv if conf_by_team.get(r["team"]) in config.P4]
    36	    dist = {k: [dig(r, *path) for r in rows] for k, path in P4_METRICS.items()}
    37	    return {
    38	        "n_teams": len(rows),
    39	        "median": {k: (round(st.median([v for v in vs if v is not None]), 4)
    40	                       if any(v is not None for v in vs) else None)
    41	                   for k, vs in dist.items()},
    42	        "dist": {k: [round(v, 4) for v in vs if v is not None] for k, vs in dist.items()},
    43	    }

    47	    team, season, week = config.TEAM, rw.season, rw.week
    48	
    49	    games = client.get("/games", year=season, team=team)

    58	                 "home_points": game["homePoints"], "away_points": game["awayPoints"],
    59	                 "excitement": game.get("excitementIndex")},
    60	        "schedule": [
    61	            {"week": g["week"],
    62	             "opponent": g["awayTeam"] if g["homeTeam"] == team else g["homeTeam"],
    63	             "home": g["homeTeam"] == team,
    64	             "pf": g["homePoints"] if g["homeTeam"] == team else g["awayPoints"],
    65	             "pa": g["awayPoints"] if g["homeTeam"] == team else g["homePoints"]}
    66	            for g in sorted(games, key=lambda z: z["week"])],
    67	    }
    68	
    69	    snap["ppa_games"] = client.get("/ppa/games", year=season, team=team)
    70	    snap["ppa_team_season"] = client.get("/ppa/teams", year=season, team=team)
    71	    snap["adv_games_baylor"] = client.get("/stats/game/advanced", year=season, team=team)
    72	    snap["adv_game"] = [r for r in snap["adv_games_baylor"] if r.get("gameId") == rw.game_id]
    73	    snap["adv_season_baylor"] = client.get("/stats/season/advanced", year=season, team=team)
    74	    snap["drives"] = client.get("/drives", year=season, week=week, team=team)
    75	    snap["plays"] = client.get("/plays", year=season, week=week, team=team)
    76	    snap["winprob_raw"] = client.get("/metrics/wp", gameId=rw.game_id)
    77	
    78	    if rw.next_opponent:
    79	        snap["ppa_next_opp"] = client.get("/ppa/teams", year=season, team=rw.next_opponent)
    80	        snap["adv_season_next_opp"] = client.get("/stats/season/advanced", year=season,
    81	                                                 team=rw.next_opponent)
    82	        snap["next_opp_ppa_games"] = client.get("/ppa/games", year=season,
    83	                                                team=rw.next_opponent)
    84	        snap["next_opp_adv_games"] = client.get("/stats/game/advanced", year=season,
    85	                                                team=rw.next_opponent)
    86	    else:
    87	        snap["ppa_next_opp"] = snap["adv_season_next_opp"] = []
    88	        snap["next_opp_ppa_games"] = snap["next_opp_adv_games"] = []
    89	
    90	    all_adv = client.get("/stats/season/advanced", year=season)
    91	    conf = {t["school"]: t.get("conference")
    92	            for t in client.get("/teams/fbs", year=season)}
    93	    snap["p4"] = p4_distributions(all_adv, conf)
    94	    snap["baylor_season_adv"] = snap["adv_season_baylor"][0] if snap["adv_season_baylor"] else {}
    95	    snap["next_opp_season_adv"] = (snap["adv_season_next_opp"][0]
    96	                                   if snap["adv_season_next_opp"] else {})
    97	
    98	    tid = pff.team_id(con)
    99	    snap["meta"]["pff_team_id"] = tid
   100	    snap["pff_grades"] = pff.game_grades(con, season, week)
   101	    snap["pff_season_snaps"] = pff.season_snap_rows(con, season, tid, through_week=week)
   102	    return snap

    16	def _out_dir(season: int, week: int):
    17	    return config.OUT / str(season) / f"wk{week:02d}"

    19	
    20	def _load_or_pull(season, week, offline):
    21	    """Return (snapshot, ResolvedWeek). One path builds, one path reloads.
    22	
    23	    The offline path reconstructs the next game from `snapshot["schedule"]`,
    24	    which is the whole season's week/opponent/home, not just the played games —
    25	    it was measured to carry this, contrary to an earlier draft's assumption
    26	    that next_week/next_home are simply unavailable offline. Only a genuine
    27	    last-game-of-season snapshot (no schedule row past the frozen week) falls
    28	    back to `next_opponent=None`, and `build_meta` degrades that to "No further
    29	    games" honestly rather than printing the literal "Week None".
    30	    """
    31	    if offline:
    32	        cached = sorted(config.OUT.glob(f"{season}/wk*/snapshot.json"))
    33	        if not cached:
    34	            raise NoCompletedGame(f"no cached snapshot for {season}")
    35	        snap = json.loads(cached[-1].read_text())
    36	        m = snap["meta"]
    37	        if week is not None and week != m["week"]:
    38	            raise NoCompletedGame(
    39	                f"--offline reloads the most recent cached snapshot, which is week "
    40	                f"{m['week']}, not the week {week} you asked for. "
    41	                f"Drop --offline to rebuild week {week} from the API.")
    42	        later = sorted((g for g in snap.get("schedule", []) if g["week"] > m["week"]),
    43	                       key=lambda g: g["week"])
    44	        nxt = later[0] if later else None
    45	        rw = ResolvedWeek(m["year"], m["week"], m["game_id"], m["opponent"],

     1	"""Gates G6 and G10 — the two rules that are easy to break by accident.
     2	
     3	G6 is the zero-hand-edits rule made checkable. G10 keeps the voice descriptive:
     4	DESIGN.md forbids tier labels, and a standing reviewer preference forbids
     5	blame-implying rationale.
     6	
     7	Four sources of false positives were MEASURED on the real templates and source
     8	tree and are stripped before scanning. Without these strips the audit reports
     9	hits on genuinely clean files, and an audit that cries wolf gets switched off:
    10	  - the embedded font CSS carries `U+2000-206F` and `U+2074`, which read as years;
    11	  - the SVG namespace `http://www.w3.org/2000/svg` reads as the year 2000;
    12	  - the design brief at the top of each template is an HTML comment and names the
    13	    week's opponent in prose. A comment is not printed, so it is not a hand-edit;
    14	  - once G6 was extended to scan `src/`, the same reasoning applies one level
    15	    down: `fonts.py`'s `unicode-range` constant repeats the `U+2000-206F` /
    16	    `U+2074` tokens outside any `<style>` block, and two module docstrings
    17	    (`payload.py`, `sources/pff.py`) mention "the 2024 report" and "2023-2025"
    18	    in prose. Neither a Python comment nor a docstring is ever printed to a
    19	    sheet, so — same rationale as the HTML-comment strip above — both are
    20	    blanked before scanning `.py` files. Only the docstring string itself and
    21	    `#` comments are blanked; every other string literal in the file (a SQL
    22	    query, an f-string, an assignment) is left untouched, so a real hardcode
    23	    sitting in live code is still caught.
    24	
    25	With these four strips, both prototype templates AND the entire `src/coachreport`
    26	tree scan CLEAN, and a file containing `<p>Week 6, 2025 vs Kansas State</p>` (or
    27	the Python equivalent `SEASON = 2025; LABEL = "Week 6"; OPP = "Kansas State"` in
    28	live code) is still caught on all three counts. All of this was verified before
    29	this plan was written and re-verified when Ruling 3 extended the scan to `src/`.

    31	import ast
    32	import io
    33	import json
    34	import re
    35	import tokenize
    36	from pathlib import Path
    37	
    38	COMMENT = re.compile(r"<!--.*?-->", re.S)
    39	FONT_STYLE = re.compile(r"<style>\s*/\* Self-hosted:.*?</style>", re.S)
    40	# Deliberately blanket-wide, not scoped to comments/docstrings: the SVG namespace
    41	# `http://www.w3.org/2000/svg` appears in LIVE template JavaScript (not a comment),
    42	# and contains the digits `2000`, which would otherwise read as a hardcoded year.
    43	# Known cost of that width: a genuine year hardcoded inside a URL in live source
    44	# (e.g. a typo'd API endpoint with a year baked in) would be silently missed. That
    45	# tradeoff was chosen deliberately rather than narrowed, because narrowing would
    46	# make the SVG namespace trip G6 on every run.
    47	URL = re.compile(r"https?://\S+")
    48	UNICODE_RANGE = re.compile(r"U\+[0-9A-Fa-f]{4}(?:-[0-9A-Fa-f]{4})?")
    49	YEAR = re.compile(r"\b(?:19|20)\d{2}\b")
    50	WEEK = re.compile(r"\bWeek\s+\d+\b", re.I)
    51	# Big 12 opponents plus the non-conference names on the 2025 card. Config-free:
    52	# any of these appearing as a literal in markup is a hand-edit waiting to happen.
    53	OPPONENTS = [
    54	    "Kansas State", "Kansas", "TCU", "Texas Tech", "Oklahoma State", "Iowa State",
    55	    "Cincinnati", "Houston", "BYU", "Utah", "Arizona", "Arizona State", "Colorado",
    56	    "UCF", "West Virginia", "Auburn", "SMU", "Samford", "Arkansas",
    57	]
    58	TIER_WORDS = ["elite", "poor", "weak", "bad", "underperform", "disappointing",
    59	              "worst", "lazy", "soft"]

    63	def _python_docstring_and_comment_spans(text: str) -> list[tuple[int, int, int, int]]:
    64	    """1-indexed-line/0-indexed-col spans of every `#` comment and every module,
    65	    class or function docstring in a Python source file. Neither is printed to
    66	    a sheet, so neither counts as a hand-edit."""
    67	    spans = []
    68	    try:
    69	        for tok in tokenize.generate_tokens(io.StringIO(text).readline):
    70	            if tok.type == tokenize.COMMENT:
    71	                (sr, sc), (er, ec) = tok.start, tok.end
    72	                spans.append((sr, sc, er, ec))
    73	    except (tokenize.TokenError, IndentationError):
    74	        pass
    75	    try:
    76	        tree = ast.parse(text)
    77	    except SyntaxError:
    78	        tree = None
    79	    if tree is not None:
    80	        doc_holders = (ast.Module, ast.ClassDef, ast.FunctionDef, ast.AsyncFunctionDef)
    81	        for node in ast.walk(tree):
    82	            if not isinstance(node, doc_holders) or not node.body:
    83	                continue
    84	            first = node.body[0]
    85	            if (isinstance(first, ast.Expr) and isinstance(first.value, ast.Constant)
    86	                    and isinstance(first.value.value, str)):
    87	                v = first.value
    88	                spans.append((v.lineno, v.col_offset, v.end_lineno, v.end_col_offset))
    89	    return spans
    90	
    91	
    92	def _blank(text: str, spans: list[tuple[int, int, int, int]]) -> str:
    93	    lines = text.splitlines(keepends=True)
    94	    for sr, sc, er, ec in spans:
    95	        if sr == er:
    96	            line = lines[sr - 1]
    97	            lines[sr - 1] = line[:sc] + " " * (ec - sc) + line[ec:]
    98	            continue
    99	        first = lines[sr - 1]
   100	        nl = "\n" if first.endswith("\n") else ""
   101	        lines[sr - 1] = first[:sc] + " " * (len(first) - sc - len(nl)) + nl
   102	        for i in range(sr, er - 1):
   103	            nl2 = "\n" if lines[i].endswith("\n") else ""
   104	            lines[i] = " " * (len(lines[i]) - len(nl2)) + nl2
   105	        last = lines[er - 1]
   106	        lines[er - 1] = " " * ec + last[ec:]
   107	    return "".join(lines)
   108	
   109	
   110	def _strip_python_noise(text: str) -> str:
   111	    return _blank(text, _python_docstring_and_comment_spans(text))

   114	def no_hardcoded_weekly_values(paths) -> list[str]:
   115	    hits = []
   116	    for p in (Path(x) for x in paths):
   117	        if p.name in EXEMPT:
   118	            continue
   119	        text = p.read_text(errors="ignore")
   120	        if p.suffix == ".py":
   121	            text = _strip_python_noise(text)
   122	        text = URL.sub("", FONT_STYLE.sub("", COMMENT.sub("", text)))
   123	        text = UNICODE_RANGE.sub("", text)
   124	        for m in YEAR.findall(text):
   125	            hits.append(f"{p.name}: hardcoded year {m}")
   126	        for m in WEEK.findall(text):
   127	            hits.append(f"{p.name}: hardcoded {m}")
   128	        for name in OPPONENTS:
   129	            if re.search(rf"\b{re.escape(name)}\b", text):
   130	                hits.append(f"{p.name}: hardcoded opponent {name}")
   131	    return sorted(set(hits))

   136	    return [w for w in TIER_WORDS if re.search(rf"\b{w}\b", text)]

     1	"""Headless Chrome export, plus the gates that decide whether a run may ship.
     2	
     3	Three traps this module exists to avoid:
     4	  - Chrome stamps a file:/// path into the footer by default. --no-pdf-header-footer.
     5	  - Byte size proves nothing. Page COUNT and page BOX are the arbiters, and a
     6	    sheet that silently spilled onto a sixth page looks fine until it is printed.
     7	  - `--evaluate-on-new-document` is a Puppeteer API, NOT a Chrome CLI flag. Chrome
     8	    ignores unknown flags in silence, so a probe written that way returns nothing
     9	    and every gate built on it passes vacuously. The probe below injects its script
    10	    into a temporary copy of the document and reads the answer back out of
    11	    <title> via --dump-dom, which does work from the command line.
    12	
    13	The /Type/Pages and /MediaBox extraction below was verified against the two
    14	known-good prototype PDFs: both report 5 pages at 8.5 x 11.0 in.
    15	"""
    16	import json
    17	import re
    18	import subprocess
    19	import tempfile
    20	from pathlib import Path
    21	
    22	CHROME = "/Applications/Google Chrome.app/Contents/MacOS/Google Chrome"
    23	OFFLINE_FLAGS = ["--host-resolver-rules=MAP * 0.0.0.0"]
    24	BASE_FLAGS = ["--headless=new", "--disable-gpu", "--no-sandbox",
    25	              "--no-pdf-header-footer", "--virtual-time-budget=8000"]

    28	def _run(args: list[str]) -> str:
    29	    r = subprocess.run(args, capture_output=True, text=True, timeout=60)
    30	    if r.returncode != 0:
    31	        raise RuntimeError(f"chrome failed: {r.stderr[-800:]}")
    32	    return r.stdout

    34	
    35	def to_pdf(html_path: Path, pdf_path: Path, offline: bool = True) -> Path:
    36	    flags = BASE_FLAGS + (OFFLINE_FLAGS if offline else [])
    37	    _run([CHROME, *flags, f"--print-to-pdf={pdf_path}", f"file://{html_path}"])
    38	    if not pdf_path.exists() or pdf_path.stat().st_size == 0:
    39	        raise RuntimeError(f"no PDF produced at {pdf_path}")
    40	    return pdf_path

    43	def page_count(pdf_path: Path) -> int:
    44	    raw = pdf_path.read_bytes()
    45	    counts = [int(m) for m in re.findall(rb"/Type\s*/Pages[^>]*?/Count\s+(\d+)", raw)]
    46	    if counts:
    47	        return max(counts)
    48	    return len(re.findall(rb"/Type\s*/Page[^s]", raw))
    49	
    50	
    51	def page_size_inches(pdf_path: Path) -> tuple[float, float]:
    52	    raw = pdf_path.read_bytes()
    53	    m = re.search(rb"/MediaBox\s*\[\s*([\d.]+)\s+([\d.]+)\s+([\d.]+)\s+([\d.]+)", raw)
    54	    if not m:
    55	        raise RuntimeError("no /MediaBox in the PDF")
    56	    x0, y0, x1, y1 = (float(v) for v in m.groups())
    57	    return round((x1 - x0) / 72, 2), round((y1 - y0) / 72, 2)
    58

    60	# Appended to a TEMPORARY copy of the document. It writes its answer into the
    61	# title, because --dump-dom is the only channel Chrome gives us from the CLI.
    62	#
    63	# NOTE: this does NOT use document.fonts.check(). In this headless Chrome build,
    64	# check() returns true for ANY family name -- including nonsense ones that were
    65	# never declared anywhere in the document -- so a font gate built on it passes
    66	# vacuously on every input, print-quality-affecting exactly like the
    67	# --evaluate-on-new-document trap the module docstring warns about, just via a
    68	# different API. Instead this walks the live document.fonts FontFaceSet and
    69	# checks, for each expected family, whether a FontFace with that family actually
    70	# reached status "loaded" -- present only when the @font-face rule was declared
    71	# AND its data actually decoded offline.
    72	_PROBE_JS = """
    73	<script>
    74	document.fonts.ready.then(() => {
    75	  const sheets = [...document.querySelectorAll('.sheet')];
    76	  const loaded = [...document.fonts]
    77	    .filter(f => f.status === 'loaded')
    78	    .map(f => f.family.replace(/^['"]|['"]$/g, ''));
    79	  document.title = 'PROBE:' + JSON.stringify({
    80	    sheets: sheets.length,
    81	    tall: sheets.map((s, i) => [i, s.scrollHeight]).filter(([, h]) => h > 1056),
    82	    fonts: ['Archivo Narrow', 'JetBrains Mono', 'Source Sans 3']
    83	             .map(f => loaded.includes(f)),
    84	    bad: (document.body.innerText.match(/undefined|NaN/g) || []).length,
    85	  });
    86	});
    87	</script>
    88	"""

    91	class ProbeFailed(RuntimeError):
    92	    """The probe did not report. Never treat this as 'no problems found'."""
    93

    95	def probe(html_path: Path) -> dict:
    96	    """Measure the rendered document with the network blackholed.
    97	
    98	    Raises rather than returning {} — a silent empty probe is how G2/G3/G4 turn
    99	    into gates that cannot fail.
   100	    """
   101	    with tempfile.TemporaryDirectory() as d:
   102	        probed = Path(d) / html_path.name
   103	        probed.write_text(html_path.read_text() + _PROBE_JS)
   104	        # Headless Chrome's default window is 756x600 -- narrower than the
   105	        # template's own "@media screen and (max-width:9in)" responsive
   106	        # breakpoint. Left at the default, the probe measures the mobile
   107	        # layout (full-width .sheet, different padding), not the one that
   108	        # actually gets printed, which makes the overflow gate (G2) both
   109	        # misreport the overflow size and potentially miss real overflow.
   110	        out = _run([CHROME, *BASE_FLAGS, *OFFLINE_FLAGS,
   111	                    "--window-size=1280,1200", "--dump-dom",
   112	                    f"file://{probed}"])
   113	    m = re.search(r"<title>PROBE:(.*?)</title>", out, re.S)
   114	    if not m:
   115	        raise ProbeFailed(f"probe produced no reading for {html_path.name}")
   116	    return json.loads(m.group(1))

   119	def check_gates(html_path: Path, pdf_path: Path) -> list[str]:
   120	    fails = []
   121	    html = html_path.read_text()
   122	
   123	    n = page_count(pdf_path)
   124	    if n != 5:
   125	        fails.append(f"G1 page count is {n}, expected 5")
   126	    w, h = page_size_inches(pdf_path)
   127	    if (w, h) != (8.5, 11.0):
   128	        fails.append(f"G1 page box is {w}x{h}in, expected 8.5x11.0")
   129	
   130	    for host in ("fonts.googleapis.com", "fonts.gstatic.com"):
   131	        if host in html:
   132	            fails.append(f"G5 output references {host}")
   133	
   134	    try:
   135	        p = probe(html_path)
   136	    except ProbeFailed as e:
   137	        return fails + [f"G2/G3/G4 could not be measured: {e}"]
   138	
   139	    if p.get("tall"):
   140	        fails.append(f"G2 sheets over one page: {p['tall']}")
   141	    if p.get("bad"):
   142	        fails.append(f"G3 {p['bad']} undefined/NaN in rendered text")
   143	    if not all(p.get("fonts", [])):
   144	        fails.append(f"G4 fonts not loaded offline: {p.get('fonts')}")
   145	    return fails

     1	"""Inject a payload into a template. The template is never edited by hand."""
     2	import json
     3	from pathlib import Path
     4	
     5	from . import config, fonts
     6	
     7	SLOT = "__PAYLOAD__"
     8	
     9	
    10	def inject(template_html: str, payload: dict) -> str:
    11	    if SLOT not in template_html:
    12	        raise ValueError(
    13	            f"no {SLOT} placeholder in template — refusing to ship a sheet with no data")
    14	    data = json.dumps(payload, separators=(",", ":"), default=str)
    15	    # keep a play description containing </script> from closing the block
    16	    data = data.replace("</", "<\\/")
    17	    return template_html.replace(SLOT, data)
    18	
    19	
    20	def render_edition(edition: str, payload: dict, out_dir: Path) -> Path:
    21	    if edition not in config.EDITIONS:
    22	        raise ValueError(f"unknown edition {edition!r}; known: {sorted(config.EDITIONS)}")
    23	    template = config.TEMPLATES / config.EDITIONS[edition]
    24	    html = fonts.embed(inject(template.read_text(encoding="utf-8"), payload))
    25	    out_dir.mkdir(parents=True, exist_ok=True)
    26	    out = out_dir / f"{edition}.html"
    27	    out.write_text(html, encoding="utf-8")
    28	    return out
