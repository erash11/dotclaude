   112	    db.execute("CREATE VIRTUAL TABLE IF NOT EXISTS d USING fts5(path UNINDEXED, name, body, tokenize='porter unicode61')")
   113	    db.execute("CREATE TABLE IF NOT EXISTS seen(path TEXT PRIMARY KEY, mtime REAL)")
   114	    db.execute("BEGIN IMMEDIATE")  # two refreshes at once could each insert the same new file

   116	    now = {}

   118	        try:
   119	            now[p] = os.stat(p).st_mtime
   120	        except OSError:
   121	            pass

   125	        db.execute("DELETE FROM d WHERE path = ?", (p,))

   131	            continue

   133	        db.execute("INSERT INTO d VALUES (?, ?, ?)", (p, name, name + "\n" + body))
   134	        db.execute("INSERT INTO seen VALUES (?, ?)", (p, now[p]))
   135	    db.execute("COMMIT")

   143	    """BM25 rank of whole files for a natural-language query (any word may match). The file name is

   183	def rank(db, q, limit):
   184	    """BM25 order, except a Tier A memory in the top 3 moves to #1 when what it says it is about
   185	    (name + description) matches as many question words as #1's file name. Compiled knowledge
   186	    articles and vault notes can be older than the memory that corrected them; the guard stops an
   187	    off-topic memory jumping. (Giving #1 its title too lost n06 on the 30-case eval.)"""
   188	    cands = [newest_copy(db, p) if tier_a(p) else p for p in fts5_top(db, q, max(limit, 3) + 3)]
   189	    cands = list(dict.fromkeys(cands))
   190	    # global-memory/handoffs keeps a .windows.md twin of many files; one copy is enough.
   191	    cands = [p for p in cands if not (p.endswith(".windows.md") and p[:-len(".windows.md")] + ".md" in cands)]
   192	    # The best-matching Tier A memory in the top 3, not the first: a superseded memory at #1 must
   193	    # not shield the current one at #2 (p01 flipped on 2026-10-24 for this reason).
   194	    tops = [p for p in cands[:3] if tier_a(p)]
   195	    a = max(tops, key=lambda p: (about_hits(db, p, q), -cands.index(p))) if tops else None
   196	    if a and a != cands[0] and about_hits(db, a, q) >= about_hits(db, cands[0], q, describe=False):
   197	        cands.remove(a)
   198	        cands.insert(0, a)
   199	    return cands[:limit]

   202	def idf_of(db, kws):
   203	    n = db.execute("SELECT count(*) FROM seen").fetchone()[0] or 1
   204	    out = {}
   205	    for k in kws:
   206	        try:
   207	            df = db.execute("SELECT count(*) FROM d WHERE d MATCH ?", ('"%s"*' % k,)).fetchone()[0]
   208	        except sqlite3.OperationalError:
   209	            df = 0
   210	        out[k] = math.log(1 + n / (1 + df))
   211	    return out

   216	def paragraphs(path):
   217	    """(first line no, heading, text) per blank-line block, frontmatter dropped except description."""
   218	    lines = open(path, encoding="utf-8", errors="replace").read().split("\n")
   219	    out, start, heading, buf = [], 0, "", []
   220	    i = 0
   221	    if lines and lines[0].strip() == "---":
   222	        for j in range(1, len(lines)):
   223	            if lines[j].strip() == "---":
   224	                desc = [l for l in lines[1:j] if l.startswith("description:")]
   225	                if desc:
   226	                    out.append((2, "", desc[0]))
   227	                i = j + 1
   228	                break
   229	    for n in range(i, len(lines) + 1):
   230	        line = lines[n] if n < len(lines) else ""
   231	        if line.strip() == "" or n == len(lines):
   232	            if buf:
   233	                out.append((start + 1, heading, "\n".join(buf)))
   234	                buf = []
   235	            continue
   236	        if line.startswith("#"):
   237	            if buf:
   238	                out.append((start + 1, heading, "\n".join(buf)))
   239	                buf = []
   240	            heading = line.lstrip("#").strip()
   241	            continue
   242	        if not buf:
   243	            start = n
   244	        buf.append(line)
   245	    return out

   248	def best_paragraphs(path, kws, idf, k=3):
   249	    paras = paragraphs(path)
   250	    if sum(len(p[2]) for p in paras) <= WHOLE:
   251	        return [(0, i, ln, h, t, {w for w in kws if w in set(tokens(h + " " + t))})
   252	                for i, (ln, h, t) in enumerate(paras)]
   253	    scored = []
   254	    for pos, (ln, head, text) in enumerate(paras):
   255	        toks = set(tokens(head + " " + text))
   256	        s = sum(idf.get(w, 1) for w in kws if w in toks)
   257	        if s:
   258	            scored.append((s, pos, ln, head, text, {w for w in kws if w in toks}))
   259	    scored.sort(key=lambda x: -x[0])
   260	    chosen, used = [], 0
   261	    for item in scored[:k]:
   262	        if chosen and used + len(item[4]) > BUDGET:
   263	            continue
   264	        chosen.append(item)
   265	        used += len(item[4])
   266	    # The opening paragraph usually states the main fact ("deployed at <url>") in words the question
   267	    # may not share, so a short one always comes along, outside the budget.
   268	    lead = next(((pos, ln, h, t) for pos, (ln, h, t) in enumerate(paras) if not t.startswith("description:")), None)
   269	    if lead and len(lead[3]) <= LEAD and lead[0] not in {c[1] for c in chosen}:
   270	        pos, ln, h, t = lead
   271	        chosen.append((0, pos, ln, h, t, {w for w in kws if w in set(tokens(h + " " + t))}))
   272	    chosen.sort(key=lambda x: x[1])  # back to document order
   273	    return chosen

   283	def pointers(text, from_path):
   284	    """Resolvable [[name]] links and backticked .md paths, in order of appearance."""
   285	    found = []
   286	    for name in re.findall(r"\[\[([^\]|#]+)", text):

   289	                     os.path.join(os.path.dirname(from_path), "project_" + name + ".md"),
   290	                     os.path.join(KNOWLEDGE, name + ".md")):

   293	                break
   294	        else:
   295	            hits = glob.glob(os.path.join(VAULT, "**", name + ".md"), recursive=True)
   296	            if hits:
   297	                found.append(hits[0])
   298	    for p in re.findall(r"`(~?/[^`\s]+\.md)`", text):
   299	        p = os.path.expanduser(p)

   305	def gather(question, db, top=3):
   306	    """Returns (kws, candidates, evidence) where evidence is [(path, [paragraph tuples])]."""
   307	    kws = keywords(question)
   308	    cands = rank(db, question, top)
   309	    if not cands or not os.path.exists(cands[0]):
   310	        return kws, cands, []
   311	    idf = idf_of(db, kws)
   312	    path = cands[0]
   313	    chosen = best_paragraphs(path, kws, idf)
   314	    evidence = [(path, chosen)]
   315	    covered = set().union(*(c[5] for c in chosen)) if chosen else set()
   316	    # One hop, gated: only when this file misses at least half the keywords.
   317	    if len(covered) * 2 < len(kws):
   318	        text = "\n".join(c[4] for c in chosen) or open(path, encoding="utf-8", errors="replace").read()
   319	        for target in pointers(text, path)[:1]:
   320	            extra = best_paragraphs(target, kws, idf)
   321	            if extra and set().union(*(c[5] for c in extra)) - covered:
   322	                evidence.append((target, extra))
   323	    return kws, cands, evidence

   330	def memory_first(hits, limit=5):
   331	    """memsearch hits -> one best chunk per file, Tier A memory files first, at most `limit` files.
   332	    Same idea as rank()'s Tier A rule. Eval 2026-10-25 (30 cases): hit@1 13 -> 16, hit@5 25 -> 27,
   333	    no vault case worse. A Windows-slug twin is dropped when the Mac copy is also listed;
   334	    _archive memories are old versions, so they do not move up."""
   335	    best = {}
   336	    for h in hits:
   337	        best.setdefault(h["source"], h)  # memsearch returns best score first
   338	    win = lambda p: "/projects/C--" in p or "/projects/c--" in p
   339	    mac_names = {os.path.basename(p) for p in best if not win(p)}
   340	    files = [p for p in best if not (win(p) and os.path.basename(p) in mac_names)]
   341	    up = lambda p: tier_a(p) and "/_archive/" not in p
   342	    return [best[p] for p in [p for p in files if up(p)] + [p for p in files if not up(p)]][:limit]

   347	    5 files shown; about 8-10 s (top 20 costs the same as top 5, measured 2026-10-25)."""

   351	                             capture_output=True, text=True, timeout=45)  # ~4x measured; a locked index must not hang a ~1 s tool
   352	    except (OSError, subprocess.TimeoutExpired) as e:
   353	        return f"memsearch did not run: {e}"
   354	    if res.stdout.strip():

   359	        out = []

   369	    last = (res.stderr.strip().splitlines() or ["no output"])[-1][:200]
